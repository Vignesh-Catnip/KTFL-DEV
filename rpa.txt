defVar --name SAP_Easy_Access --type Window
defVar --name SAP_Easy_Access_success --type Boolean
defVar --name sap_logon_window --type Window
defVar --name sap_logon_window_success --type Boolean
defVar --name login_window --type Window
defVar --name login_window_success --type Boolean

defVar --name username --type String
defVar --name password --type String

defVar --name sap_prd_user --type String
defVar --name sap_prd_pass --type String
defVar --name ktfl_sap_user --type String
defVar --name ktfl_sap_pass --type String
defVar --name T_code --type String
defVar --name company_code_window --type Window
defVar --name company_code_windows_Success --type Boolean
defVar --name company_code --type String
defVar --name db_connection --type DbConnection
defVar --name invoice_table --type DataTable
defVar --name total_rows --type Numeric
defVar --name current_row --type Numeric
defVar --name invoice_number --type String
defVar --name invoice_date --type String
defVar --name po_number --type String

getParameter --name ktfl_sap_user ktfl_sap_user=value
getParameter --name "ktfl-sap-pass" ktfl_sap_pass=value

setVar --name "${username}" --value "${ktfl_sap_user}"
setVar --name "${password}" --value "${ktfl_sap_pass}"


// Check whether SAP is already logged in
waitWindow --title "Enter Incoming Invoice: Company Code 2000" --classname SAP_FRONTEND_SESSION --processname saplogon --timeout "00:00:05" SAP_Easy_Access=value SAP_Easy_Access_success=success

if --left "${SAP_Easy_Access_success}" --operator "Is_True" --negate
	// Check SAP Logon window
	findWindow --title "SAP Logon 800" sap_logon_window=value sap_logon_window_success=success
	// If SAP Logon is not open, launch it
	if --left "${sap_logon_window_success}" --operator "Is_True" --negate
		powerShell --apartmentState "MTA" --script "\"Start-Process"
		waitWindow --title "SAP Logon 800" --classname "#32770" --processname saplogon --timeout "00:00:10" sap_logon_window=value sap_logon_window_success=success
	endIf
	// Attach SAP Logon
	attachWindow --window ${sap_logon_window}
	// Maximize SAP Logon
	showWindow --maximize
	// Select KTFL SAP connection
	click --selector "XPath" --controlsimilarity 100 --xpath "/root/pane[1]/pane[1]/pane[1]/pane[1]/pane[1]/pane[2]/pane[1]/list[1]/listitem[1]/text[1]"
	delay --timeout "00:00:00.5000000"
	// Open selected connection
	keyboard --type "KeyDown" --key "Return"
	// Wait for SAP login screen
	waitWindow --title SAP --classname SAP_FRONTEND_SESSION --processname saplogon --timeout "00:00:15" login_window=value login_window_success=success
	if --left "${login_window_success}" --operator "Is_True"
		attachWindow --window ${login_window}
		// Username
		typeText --text "${username}"
		delay --timeout "00:00:00.5000000"
		// Move to Password field
		keyboard --type "KeyDown" --key "Tab"
		delay --timeout "00:00:00.5000000"
		// Password
		typeText --text "${password}"
		delay --timeout "00:00:00.5000000"
		// Login
		keyboard --type "KeyDown" --key "Return"
		// Wait for SAP Easy Access
		waitWindow --title "SAP Easy Access  -  User Menu for Abhijeet Chaudhari" --classname SAP_FRONTEND_SESSION --processname saplogon --safesearch  --timeout "00:00:15" SAP_Easy_Access=value SAP_Easy_Access_success=success
	else
		logMessage --message "SAP Login window failed to appear." --type "Warning"
	endIf
	
endIf


// Check login success
if --left "${SAP_Easy_Access_success}" --operator "Is_True"
	logMessage --message "SAP login successful. T-Code box selected." --type "Info"
	attachWindow --window ${SAP_Easy_Access}
	// Click T-Code box
	click --selector "XPath" --controlsimilarity 100 --xpath "/root/pane[2]/combobox[1]"
	setVar --name "${T_code}" --value MIRO
	typeText --text "${T_code}"
	keyboard --type "KeyPress" --key "Return"
	waitWindow --title "Enter Company Code" --classname "#32770" --processname saplogon --safesearch  --timeout "00:00:03" company_code_window=value company_code_windows_Success=success
	if --left "${company_code_windows_Success}" --operator "Is_True"
		attachWindow --window ${company_code_window}
		setVar --name "${company_code}" --value 2000
		typeText --text "${company_code}"
		delay --timeout "00:00:03"
		keyboard --type "KeyPress" --key "Return"
	else
		logMessage --message "company code window not showing" --type "Info"
	endIf
	
	
else
	
	logMessage --message "SAP login failed. SAP Easy Access screen not found." --type "Error"
	
endIf

logMessage --message "SAP Login process completed." --type "Info"

postgreConnect --connectionstring "Host=10.7.11.22;Port=5432;Database=miro_invoice;User ID=appuser;Password=appuser123;" db_connection=connection
sqlExecuteReader --connection ${db_connection} --statement "SELECT\r\n    id,\r\n    invoice_number,\r\n    invoice_date,\r\n    po_number\r\nFROM invoices\r\nWHERE status = \'REVIEW\'\r\nORDER BY id" invoice_table=value total_rows=rows
for --variable ${current_row} --from 1 --to ${total_rows} --step 1
	mapTableRow --dataTable ${invoice_table} --row ${current_row} --mappings "[{\"Number\":\"2\",\"Output\":\"${invoice_number}\"},{\"Number\":\"3\",\"Output\":\"${invoice_date}\"},{\"Number\":\"4\",\"Output\":\"${po_number}\"}]"
	logMessage --message "${invoice_number}\r\n${invoice_date}\r\n${po_number}" --type "Info"
	waitWindow --title "Enter Incoming Invoice: Company Code 2000" --classname SAP_FRONTEND_SESSION --processname saplogon --safesearch
next
